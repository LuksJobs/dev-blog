{{/* Carousel of posts on X (params.xFeed.posts = list of post URLs, newest first). Each post is fetched at build time from X's public syndication endpoint; posts that can't be fetched are skipped with a warning. */}} {{- $posts := slice -}} {{- range .Site.Params.xFeed.posts -}} {{- $url := . -}} {{- $id := replaceRE `.*/status/(\d+).*` "$1" $url -}} {{- $api := printf "https://cdn.syndication.twimg.com/tweet-result?id=%s&token=4" $id -}} {{- with try (resources.GetRemote $api) -}} {{- with .Err -}} {{- warnf "xFeed: could not fetch %s: %s" $url . -}} {{- else with .Value -}} {{- $data := . | transform.Unmarshal -}} {{- if $data.text -}} {{- $media := "" -}} {{- $video := false -}} {{- with index ($data.mediaDetails | default slice) 0 -}} {{- $media = .media_url_https -}} {{- $video = in (slice "video" "animated_gif") .type -}} {{- end -}} {{- $posts = $posts | append (dict "url" $url "text" (replaceRE `(\s*https://t\.co/\w+)+\s*$` "" $data.text | htmlUnescape) "date" $data.created_at "name" $data.user.name "handle" $data.user.screen_name "avatar" (replace $data.user.profile_image_url_https "_normal." "_bigger.") "media" $media "video" $video ) -}} {{- else -}} {{- warnf "xFeed: post unavailable (deleted or private): %s" $url -}} {{- end -}} {{- else -}} {{- warnf "xFeed: post not found: %s" $url -}} {{- end -}} {{- end -}} {{- end -}} {{- $total := len $posts -}} {{- if $total -}}
{{- range $i, $p := $posts -}}
{{ $p.name }} @{{ $p.handle }} · {{ T "xPost.follow" }}

{{ $p.text }}

{{ with $p.media }} {{ if $p.video }} {{ T "xPost.watch" }} {{ end }} {{ end }} {{ time.Format ":date_medium" $p.date }} {{ T "xPost.readMore" }}
{{- end -}}
{{ if gt $total 1 }}
{{ range $i, $p := $posts }} {{ end }}
{{ end }}
{{- end -}}
